public int maxProfit(int[] prices) {
 	if (prices.length == 1) return 0;
	int maxProfit = 0, smallestStock = prices[0];
	for (int i = 1; i < prices.length; i++) {
		maxProfit = Math.max(maxProfit, prices[i] - smallestStock);
		smallestStock = Math.min(smallestStock, prices[i]);
	}
	return maxProfit;
}
