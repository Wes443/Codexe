public int pivotIndex(List<Integer> nums){
	int r_sum = nums.stream().mapToInt(Integer::intValue).sum();
	int l_sum = 0;
	
	for(int i = 0; i < nums.size(); i++){
		if(i == 0) r_sum -= nums.get(i);
		else{
			l_sum += nums.get(i - 1);
			r_sum -= nums.get(i);
		}
		if(l_sum == r_sum) return i;
	}
	return -1;
}
