def maxProfit(prices):
	if len(prices) == 1:
		return 0
	max_profit = 0
	smallest_stock = prices[0]
	for i in range(1, len(prices)):
		if prices[i] - smallest_stock > max_profit:
			max_profit = prices[i] - smallest_stock
		if prices[i] < smallest_stock:
			smallest_stock = prices[i]
	return max_profit
